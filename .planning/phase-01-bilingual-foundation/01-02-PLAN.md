    **A. Create `src/lib/stores.ts`** with two Zustand stores:

    ```typescript
    import { create } from "zustand";
    import { persist } from "zustand/middleware";

    // Locale store — persisted to localStorage for offline reliability
    type LocaleStore = {
      locale: "en" | "ms";
      setLocale: (locale: "en" | "ms") => void;
    };

    export const useLocaleStore = create<LocaleStore>()(
      persist(
        (set) => ({
          locale: "ms", // default Malay per UI-SPEC §4.1
          setLocale: (locale) => set({ locale }),
        }),
        { name: "luthsplayground-locale" }
      )
    );

    // Sync status store — transient UI state (not persisted)
    type SyncStore = {
      syncState: "saved" | "saving" | "offline" | "migrating";
      syncVersion: number;
      setSyncState: (state: SyncStore["syncState"]) => void;
      setSyncVersion: (v: number) => void;
    };

    export const useSyncStore = create<SyncStore>((set) => ({
      syncState: "offline",
      syncVersion: 1,
      setSyncState: (syncState) => set({ syncState }),
      setSyncVersion: (syncVersion) => set({ syncVersion }),
    }));
    ```

    Add the LanguageSwitcher function BEFORE the `export default function LearningApp()`:
    ```typescript
    function LanguageSwitcher() {
      const { locale, setLocale } = useLocaleStore();
      return (
        <div className="lang-switcher" role="group" aria-label="Switch language">
          <button
            className={`lang-option ${locale === "en" ? "active-en" : "dimmed"}`}
            onClick={() => setLocale("en")}
            aria-pressed={locale === "en"}
            aria-label="Switch to English"
          >
            <span className="lang-switcher-label">EN</span>
            <span aria-hidden="true">🇬🇧</span>
          </button>
          <button
            className={`lang-option ${locale === "ms" ? "active-ms" : "dimmed"}`}
            onClick={() => setLocale("ms")}
            aria-pressed={locale === "ms"}
            aria-label="Switch to Malay"
          >
            <span className="lang-switcher-label">MS</span>
            <span aria-hidden="true">🇲🇾</span>
          </button>
        </div>
      );
    }
    ```
