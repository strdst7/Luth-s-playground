    ```typescript
    import Dexie, { type EntityTable } from "dexie";

    // Re-export types from learning-data for consistency
    import type { SubjectId } from "@/app/learning-data";

    // Database types matching existing Progress/ParentSettings structure
    export type ProgressRecord = {
      id?: number;
      xp: number;
      lessonsCompleted: number;
      correctAnswers: number;
      totalAnswers: number;
      streak: number;
      longestStreak: number;
      badges: string[];
      subjectProgress: Record<SubjectId, { completed: number; correct: number; total: number }>;
      weeklyActivity: number[];
      lastPlayed: string;
    };

    export type SettingsRecord = {
      id?: number;
      difficulty: "Easy" | "Medium" | "Hard";
      dailyLimit: number;
      soundEnabled: boolean;
      enabledSubjects: Record<SubjectId, boolean>;
    };

    export type PlayerRecord = {
      id?: number;
      name: string;
      avatar: string;
      deviceId: string;
    };

    export type SyncQueueRecord = {
      id?: number;
      operation: string;
      data: string;
      createdAt: string;
    };

    // Dexie database class
    class LuthDatabase extends Dexie {
      progress!: EntityTable<ProgressRecord, "id">;
      settings!: EntityTable<SettingsRecord, "id">;
      player!: EntityTable<PlayerRecord, "id">;
      sync_queue!: EntityTable<SyncQueueRecord, "id">;

      constructor() {
        super("luthsplayground");
        this.version(1).stores({
          progress: "++id",
          settings: "++id",
          player: "++id",
          sync_queue: "++id, createdAt",
        });
      }
    }

    export const db = new LuthDatabase();

    // Migration from localStorage to Dexie
    export type MigrationResult = {
      success: boolean;
      recordsMigrated: number;
      error?: string;
    };

    export async function migrateFromLocalStorage(): Promise<MigrationResult> {
      const recordsMigrated = 0;
      const errors: string[] = [];

      try {
        // 1. Migrate progress
        const savedProgress = localStorage.getItem("luthsplayground-progress") || localStorage.getItem("blockbright-progress");
        if (savedProgress) {
          const parsed = JSON.parse(savedProgress) as ProgressRecord;
          await db.progress.add(parsed);
          recordsMigrated++;
        }

        // 2. Migrate settings
        const savedSettings = localStorage.getItem("luthsplayground-settings") || localStorage.getItem("blockbright-settings");
        if (savedSettings) {
          const parsed = JSON.parse(savedSettings) as SettingsRecord;
          await db.settings.add(parsed);
          recordsMigrated++;
        }

        // 3. Migrate player data
        const savedName = localStorage.getItem("luthsplayground-name") || localStorage.getItem("blockbright-name");
        const savedAvatar = localStorage.getItem("luthsplayground-avatar") || localStorage.getItem("blockbright-avatar");
        const savedDevice = localStorage.getItem("luthsplayground-device") || localStorage.getItem("blockbright-device");
        await db.player.add({
          name: savedName || "Luth",
          avatar: savedAvatar || "🧑‍🚀",
          deviceId: savedDevice || crypto.randomUUID(),
        });
        recordsMigrated++;

        // 4. Store migration flag in Dexie
        await db.settings.add({ _migrated: true } as any);

        return { success: true, recordsMigrated };
      } catch (error) {
        return {
          success: false,
          recordsMigrated,
          error: error instanceof Error ? error.message : "Unknown migration error",
        };
      }
    }

    // Helper to check if migration is needed
    export async function isMigrationNeeded(): Promise<boolean> {
      const count = await db.progress.count();
      if (count > 0) return false; // Already has data
      // Check if localStorage has data to migrate
      const hasLocalData = !!(
        localStorage.getItem("luthsplayground-progress") ||
        localStorage.getItem("blockbright-progress")
      );
      return hasLocalData;
    }
    ```

    ```typescript
    useEffect(() => {
      if (!hydrated) return;
      const runMigration = async () => {
        try {
          const needed = await isMigrationNeeded();
          if (!needed) return;
          
          setMigrationState("scanning");
          setSyncState("migrating");
          
          const result = await migrateFromLocalStorage();
          
          if (result.success) {
            setMigrationState("complete");
            setMigrationProgress(100);
            // Auto-dismiss after 1.5s
            window.setTimeout(() => {
              setMigrationState("idle");
              setSyncState(navigator.onLine ? "saved" : "offline");
            }, 1500);
          } else {
            throw new Error(result.error || "Migration failed");
          }
        } catch {
          if (migrationRetryCount < maxRetries) {
            setMigrationRetryCount((c) => c + 1);
            // Re-try after 500ms
            window.setTimeout(() => runMigration(), 500);
          } else {
            setMigrationState("error");
            setSyncState(navigator.onLine ? "saved" : "offline");
            // Keep localStorage as backup — no data loss
          }
        }
      };
      void runMigration();
    }, [hydrated, migrationRetryCount]);
    ```

    ```typescript
    function MigrationProgressIndicator({
      state, progress, onRetry,
    }: {
      state: "idle" | "scanning" | "migrating" | "complete" | "error";
      progress: number;
      onRetry: () => void;
    }) {
      const t = useTranslations();
      if (state === "idle") return null;
      
      const config = {
        scanning: { icon: "🔄", spin: true, title: t("migration_scanning"), hasBar: false },
        migrating: { icon: "🔄", spin: true, title: t("migration_migrating"), hasBar: true },
        complete: { icon: "✅", spin: false, title: t("migration_complete"), hasBar: true },
        error: { icon: "⚠️", spin: false, title: t("migration_error"), hasBar: false },
      } as const;
      
      const c = config[state];
      
      return (
        <div className="migration-overlay" role="alertdialog" aria-describedby="migration-status">
          <div className="migration-card">
            <span className={`migration-icon ${c.spin ? "spin" : ""}`}>{c.icon}</span>
            <h2 id="migration-status">{c.title}</h2>
            {c.hasBar && (
              <>
                <div className="migration-bar">
                  <i style={{ width: `${progress}%` }} />
                </div>
                <span className="migration-percent">{progress}%</span>
              </>
            )}
            {!c.hasBar && state === "scanning" && (
              <div className="migration-bar indeterminate"><i /></div>
            )}
            {state === "error" && (
              <button className="migration-retry-btn" onClick={onRetry}>
                {t("migration_retry") || "Tap to retry"} 🔄
              </button>
            )}
          </div>
        </div>
      );
    }
    ```
